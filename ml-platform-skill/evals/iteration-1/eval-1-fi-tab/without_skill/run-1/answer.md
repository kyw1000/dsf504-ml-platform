```python
# ── STEP 2 – keep the three existing tab blocks unchanged, then add tab_fi ────

with tab_metrics:
    st.markdown(

    )
    page_model_performance(uc_key, standalone=False)

with tab_curves:
    st.info(

        "reports folder. Open the **Script paths** expander on the Run Pipeline "
        "page to confirm the script has been executed."
    )

with tab_thresh:
    st.info(

        "is found during Step 5 and saved as `lgbm_optimal_threshold.txt`. "
        "Adjust it here once the champion model has been trained."
    )

# ── NEW TAB ───────────────────────────────────────────────────────────────────
with tab_fi:
    import joblib
    import pandas as pd
    import plotly.express as px
    from pathlib import Path

    st.markdown(
        "<p style='color:#888;font-size:0.88rem;margin-bottom:12px;'>"
        "Top 20 feature importances from the LightGBM champion model "
        "(gain-based, normalised). Features are ordered from highest "
        "to lowest importance.</p>",
        unsafe_allow_html=True,
    )

    model_dir = MODELS_DIR / uc["model_dir"]
    champion_path = model_dir / uc["champion"] if uc.get("champion") else None

    if champion_path is None:
        st.warning(
            "No champion model is configured for this use case. "
            "Run **Step 5 — HP Tuning + Final Training** first."
        )
    else:
        champion = load_model(str(champion_path))

        if champion is None:
            st.warning(
                f"Champion model not found at `{champion_path}`. "
                "Run **Step 5 — HP Tuning + Final Training** to generate it."
            )
        else:
            # --- Extract feature importances from the LightGBM booster -------
            # Support both a raw LGBMClassifier and a sklearn Pipeline whose
            # last step is named 'clf' or is an LGBMClassifier.
            try:
                import lightgbm as lgb

                # Unwrap a Pipeline if needed
                clf = champion
                if hasattr(champion, "named_steps"):
                    # Pipeline: grab the last estimator
                    last_step_name = list(champion.named_steps.keys())[-1]
                    clf = champion.named_steps[last_step_name]

                booster = clf.booster_  # underlying lgb.Booster

                feature_names = booster.feature_name()
                importances   = booster.feature_importance(importance_type="gain")

                fi_df = (
                    pd.DataFrame({"Feature": feature_names, "Importance": importances})
                    .sort_values("Importance", ascending=False)
                    .head(20)
                    .reset_index(drop=True)
                )

                # Normalise so bars sum to 100 % (easier to read)
                fi_df["Importance (%)"] = (
                    fi_df["Importance"] / fi_df["Importance"].sum() * 100
                ).round(2)

                # Sort ascending so the longest bar is at the top of the chart
                fi_df = fi_df.sort_values("Importance (%)", ascending=True)

                fig = px.bar(
                    fi_df,
                    x="Importance (%)",
                    y="Feature",
                    orientation="h",
                    title="Top 20 Feature Importances — LightGBM Champion (Gain)",
                    labels={"Importance (%)": "Importance (% of total gain)"},
                    text="Importance (%)",
                    color="Importance (%)",
                    color_continuous_scale="Blues",
                )
                fig.update_traces(texttemplate="%{text:.1f}%", textposition="outside")
                fig.update_layout(
                    height=560,
                    coloraxis_showscale=False,
                    yaxis_title="",
                    xaxis_title="Importance (% of total gain)",
                    margin=dict(l=10, r=40, t=50, b=40),
                )

                st.plotly_chart(fig, use_container_width=True)

                with st.expander("📋 Raw importance values"):
                    st.dataframe(
                        fi_df[["Feature", "Importance", "Importance (%)"]
                              ].sort_values("Importance (%)", ascending=False)
                               .reset_index(drop=True),
                        use_container_width=True,
                    )

            except AttributeError:
                st.error(
                    "Could not extract feature importances. "
                    "Ensure the champion model is a LightGBM estimator "
                    "(LGBMClassifier or a Pipeline ending with one)."
                )
            except Exception as exc:
                st.error(f"Unexpected error loading feature importances: {exc}")
```
